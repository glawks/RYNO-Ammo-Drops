cmake_minimum_required(VERSION 3.20)
project(ryno_ammo_drops LANGUAGES C)
if(NOT OpenRACMod_DIR)
  if(DEFINED OPENRAC_SDK)
    set(OpenRACMod_DIR "${OPENRAC_SDK}/cmake")
  else()
    file(READ "${CMAKE_CURRENT_SOURCE_DIR}/sdk.lock" lock)
    string(JSON sdk_version GET "${lock}" version)
    set(OpenRACMod_DIR "${CMAKE_CURRENT_SOURCE_DIR}/.sdk/openrac-mod-sdk-${sdk_version}/cmake")
    if(NOT EXISTS "${OpenRACMod_DIR}/OpenRACModConfig.cmake")
      message(FATAL_ERROR "SDK missing: run python3 tools/fetch_sdk.py first")
    endif()
  endif()
endif()
find_package(OpenRACMod 1.11 REQUIRED)
include(CTest)
openrac_add_mod(ryno_ammo_drops MANIFEST openrac-mod.json SOURCES src/ryno_ammo_drops.c)
target_include_directories(ryno_ammo_drops PRIVATE src)
set_target_properties(ryno_ammo_drops PROPERTIES C_STANDARD 11 C_STANDARD_REQUIRED ON)
if(NOT MSVC)
  target_compile_options(ryno_ammo_drops PRIVATE -Wall -Wextra)
endif()
if(BUILD_TESTING)
  add_executable(ryno_priority_test tests/priority_test.c)
  target_include_directories(ryno_priority_test PRIVATE src)
  add_test(NAME ryno_priority COMMAND ryno_priority_test)
endif()
